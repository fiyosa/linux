# ============================================

# ============================================
# Key bindings
# ============================================
bindkey '^[[C' forward-word         # Ctrl+Right
bindkey '^[[D' backward-word        # Ctrl+Left
bindkey '^[OC' forward-char         # Right
bindkey '^[OD' backward-char        # left
bindkey '^[[1~' beginning-of-line   # Home
bindkey '^[[4~' end-of-line         # End

# ============================================
# History configuration
# ============================================
HISTFILE=~/.zsh_history
HISTSIZE=10000
SAVEHIST=10000
setopt SHARE_HISTORY
setopt HIST_IGNORE_DUPS

# ============================================
# fzf integrasi
# ============================================
source <(fzf --zsh)
# ctrl+r show history commands
# ctrl+t show file directory currently
# ** then Tab for completions all file in this directory

# ============================================

# ============================================
# Yazi file manager, tff = 49 (ttf-jetbrains-mono-nerd)
# ============================================
function y() {
    local tmp cwd
    tmp=$(mktemp -t "yazi-cwd.XXXXXX") || return 1
    yazi "$@" --cwd-file="$tmp"
    if cwd=$(cat -- "$tmp" 2>/dev/null) && [[ -n "$cwd" && "$cwd" != "$PWD" ]]; then
        cd -- "$cwd" || echo "Failed to change directory" >&2
    fi
    rm -f -- "$tmp"
}

# ============================================

# check duplicate => echo $PATH | tr ':' '\n' | sort | uniq -d
# ============================================
